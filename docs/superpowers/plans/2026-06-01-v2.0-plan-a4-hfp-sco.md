# v2.0 Plan A.4 — HFP + SCO File Loopback Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship HFP HF + AG roles on top of v1.0 RFCOMM, including the complete Service Level Connection (BRSF/BAC/CIND/CMER), call state machine, **and a full SCO data path that runs end-to-end on `Stack.virtual()`**: a WAV file is read → mSBC/CVSD encoded → injected into the HCI SCO Data Packet stream → decoded on the peer → appended to an output WAV file.

**Architecture:** Two roles share one AT-command parser/formatter + one SLC state machine + one call state machine. The SCO data path is a separate pipeline that opens after SLC succeeds: `Setup_Synchronous_Connection` HCI command → `Synchronous_Connection_Complete` event on both sides → per-direction worker loop coupling `wave`-module file I/O to `HCISCOData` packets through `pybluehost.audio.codec.CVSDEncoder`/`MSBCEncoder` from Plan A.1. `VirtualClassicLink` (already extended in A.2/A.3 for ACL+L2CAP+RFCOMM) gains three more bridges: RFCOMM AT-command pass-through, HCI SCO link signaling, and HCI SCO Data Packet forwarding. **No OS audio device integration** (sounddevice) — that's Plan A.6.

**Tech Stack:** Python 3.10+, asyncio, stdlib `wave` for file I/O, `pybluehost.audio.codec.{CVSDEncoder,CVSDDecoder,MSBCEncoder,MSBCDecoder}` from A.1, v1.0 RFCOMM + HCI, pytest with `asyncio_mode=auto`. No new third-party deps.

- **No OS audio**: file-based I/O only (`--wav-in=in.wav --wav-out=out.wav`). sounddevice integration is Plan A.6.

HFP AT commands are CRLF-terminated lines on the RFCOMM channel. The parser/formatter is the foundation for everything else.

Three message kinds (HFP v1.7.2 §4.34 + ETSI TS 127.007):
- **Command**: `AT+CMD=arg1,arg2,...\r\n` (HF→AG) or basic `ATA\r\n` / `AT+CHUP\r\n` (no args)
- **Unsolicited response**: `\r\n+CMD: arg1,arg2,...\r\n` (AG→HF, e.g. `\r\n+CIND: 1,0,0\r\n`)
- **Final result**: `\r\nOK\r\n` or `\r\nERROR\r\n` (terminates a command exchange)

```python
"""AT command parser + formatter for HFP/HSP (HFP v1.7.2 §4.34 / ETSI TS 127.007)."""
from __future__ import annotations

import re
from dataclasses import dataclass, field


class ATParseError(ValueError):
    """Raised when an AT line cannot be parsed."""


@dataclass(frozen=True)
class ATCommand:
    """An AT command: AT<name>[?][=args]\\r\\n."""
    name: str                     # e.g. "+BRSF", "A", "+CHUP"
    args: list[str] = field(default_factory=list)
    is_query: bool = False        # "?" suffix (e.g. AT+CIND?)


@dataclass(frozen=True)
class ATResponse:
    """An unsolicited AT response: \\r\\n[+]<name>[: args]\\r\\n."""
    name: str                     # e.g. "+CIND", "RING", "+CLIP"
    args: list[str] = field(default_factory=list)


@dataclass(frozen=True)
class ATResult:
    """The final OK/ERROR terminator of a command exchange."""
    ok: bool


def parse_at_line(line: str) -> ATCommand | ATResponse | ATResult:
    if not line.endswith("\r\n"):
        raise ATParseError("AT line must end with CRLF")
    body = line[:-2]

    # Result codes
    if body.startswith("\r\n"):
        body = body[2:]
    if body == "OK":
        return ATResult(ok=True)
    if body == "ERROR":
        return ATResult(ok=False)

    # Command: starts with "AT"
    if body.startswith("AT"):
        rest = body[2:]
        # Basic command (single letter, optionally with args), or "+CMD"
        if rest.startswith("+"):
            m = re.match(r"^(\+[A-Z]+)(\?)?(?:=(.*))?$", rest)
            if not m:
                raise ATParseError(f"malformed AT command: {body!r}")
            name = m.group(1)
            is_query = m.group(2) == "?"
            args_str = m.group(3)
            args = _split_args(args_str) if args_str else []
            return ATCommand(name=name, args=args, is_query=is_query)
        else:
            # Basic command (e.g. "A" for ATA, "Z" for ATZ)
            m = re.match(r"^([A-Z])(.*)$", rest)
            if not m:
                raise ATParseError(f"malformed basic AT command: {body!r}")
            name = m.group(1)
            tail = m.group(2)
            args = _split_args(tail) if tail else []
            return ATCommand(name=name, args=args, is_query=False)

    # Unsolicited response: "+CMD: args" or "RING" / "CONNECT" etc.
    if body.startswith("+"):
        m = re.match(r"^(\+[A-Z]+)(?::\s*(.*))?$", body)
        if not m:
            raise ATParseError(f"malformed AT response: {body!r}")
        name = m.group(1)
        args_str = m.group(2)
        args = _split_args(args_str) if args_str else []
        return ATResponse(name=name, args=args)

    # Bare keyword response (RING / CONNECT / BUSY / NO CARRIER ...)
    return ATResponse(name=body, args=[])


def _split_args(s: str) -> list[str]:
    """Split AT args by commas, but respect double-quoted strings."""
    args: list[str] = []
    buf: list[str] = []
    in_quotes = False
    for ch in s:
        if ch == '"':
            in_quotes = not in_quotes
            continue
        if ch == "," and not in_quotes:
            args.append("".join(buf).strip())
            buf = []
            continue
        buf.append(ch)
    args.append("".join(buf).strip())
    return args


def format_at_command(cmd: ATCommand) -> bytes:
    """Render an ATCommand back to its on-wire form."""
    if cmd.name.startswith("+"):
        if cmd.is_query and not cmd.args:
            line = f"AT{cmd.name}?\r\n"
        elif cmd.args:
            line = f"AT{cmd.name}={','.join(cmd.args)}\r\n"
        else:
            line = f"AT{cmd.name}\r\n"
    else:
        # Basic command (single letter)
        line = f"AT{cmd.name}{''.join(cmd.args)}\r\n"
    return line.encode("ascii")


def format_at_response(resp: ATResponse) -> bytes:
    """Render an unsolicited response on-wire (with leading + trailing CRLF)."""
    if resp.args:
        body = f"{resp.name}: {','.join(resp.args)}"
    else:
        body = resp.name
    return f"\r\n{body}\r\n".encode("ascii")


def format_at_result(result: ATResult) -> bytes:
    return b"\r\nOK\r\n" if result.ok else b"\r\nERROR\r\n"
```

```python
"""HFP v1.7.2 Hands-Free Profile — Plan A.4.

Two roles: HFPClient (HF) + HFPServer (AG). Both share the AT parser, SLC state
machine, and call state machine. SCO data path uses HCI SCO Data Packets +
codec from Plan A.1 (CVSDEncoder / MSBCEncoder).
"""
from __future__ import annotations

from dataclasses import dataclass, field


# HF feature bitmask (HFP v1.7.2 §4.34.1 Table 4.6).
HF_FEATURE_EC_NR = 0x0001               # Echo cancellation / noise reduction
HF_FEATURE_THREE_WAY = 0x0002
HF_FEATURE_CLI = 0x0004                  # Caller line identification
HF_FEATURE_VOICE_RECOGNITION = 0x0008
HF_FEATURE_VOLUME_CONTROL = 0x0010
HF_FEATURE_ENHANCED_CALL_STATUS = 0x0020
HF_FEATURE_ENHANCED_CALL_CONTROL = 0x0040
HF_FEATURE_CODEC_NEGOTIATION = 0x0080
HF_FEATURE_HF_INDICATORS = 0x0100
HF_FEATURE_ESCO_S4 = 0x0200
HF_FEATURE_EVR_ACTIVATION = 0x0400
HF_FEATURE_VR_TEXT = 0x0800
HF_FEATURE_WIDEBAND_SPEECH = HF_FEATURE_CODEC_NEGOTIATION   # implied; see §4.11

# AG feature bitmask (HFP v1.7.2 §4.34.1 Table 4.7).
AG_FEATURE_THREE_WAY = 0x0001
AG_FEATURE_EC_NR = 0x0002
AG_FEATURE_VR = 0x0004
AG_FEATURE_INBAND_RINGTONE = 0x0008
AG_FEATURE_VOICE_TAG = 0x0010
AG_FEATURE_REJECT_CALL = 0x0020
AG_FEATURE_ENHANCED_CALL_STATUS = 0x0040
AG_FEATURE_ENHANCED_CALL_CONTROL = 0x0080
AG_FEATURE_EXTENDED_ERROR = 0x0100
AG_FEATURE_CODEC_NEGOTIATION = 0x0200
AG_FEATURE_HF_INDICATORS = 0x0400
AG_FEATURE_ESCO_S4 = 0x0800
AG_FEATURE_CLI = 0x1000                  # caller line identification
AG_FEATURE_WIDEBAND_SPEECH = AG_FEATURE_CODEC_NEGOTIATION   # implied; see §4.11

# Codec IDs (HFP v1.7.2 §4.11).
HF_CODEC_CVSD = 1
HF_CODEC_MSBC = 2


@dataclass
class HFPConfig:
    """Per-role configuration: feature mask + advertised codecs."""
    features: int
    codecs: list[int] = field(default_factory=lambda: [HF_CODEC_CVSD, HF_CODEC_MSBC])

    @classmethod
    def default_hf(cls) -> "HFPConfig":
        return cls(
            features=(
                HF_FEATURE_EC_NR | HF_FEATURE_CLI | HF_FEATURE_VOLUME_CONTROL
                | HF_FEATURE_CODEC_NEGOTIATION
            ),
            codecs=[HF_CODEC_CVSD, HF_CODEC_MSBC],
        )

    @classmethod
    def default_ag(cls) -> "HFPConfig":
        return cls(
            features=(
                AG_FEATURE_CLI | AG_FEATURE_REJECT_CALL
                | AG_FEATURE_CODEC_NEGOTIATION | AG_FEATURE_INBAND_RINGTONE
            ),
            codecs=[HF_CODEC_CVSD, HF_CODEC_MSBC],
        )


def select_codec(*, hf_codecs: list[int], ag_codecs: list[int]) -> int:
    """Pick the highest-preference codec supported by both sides.

    Preference: mSBC (wide-band) > CVSD (narrow-band, mandatory).
    """
    hf_set = set(hf_codecs)
    ag_set = set(ag_codecs)
    common = hf_set & ag_set
    if HF_CODEC_MSBC in common:
        return HF_CODEC_MSBC
    if HF_CODEC_CVSD in common:
        return HF_CODEC_CVSD
    raise ValueError(f"no common codec between HF {hf_codecs} and AG {ag_codecs}")
```

```python
"""HFP state machines: SLC (Service Level Connection) + Call."""
from __future__ import annotations

from enum import Enum

from pybluehost.core.statemachine import StateMachine


class SLCState(Enum):
    INITIAL = "initial"
    BRSF_DONE = "brsf_done"
    BAC_DONE = "bac_done"
    CIND_DONE = "cind_done"
    CONNECTED = "connected"
    DISCONNECTED = "disconnected"


class SLCEvent(Enum):
    # HF-side outgoing
    SEND_BRSF = "send_brsf"
    SEND_BAC = "send_bac"
    SEND_CIND_TEST = "send_cind_test"
    SEND_CIND_GET = "send_cind_get"
    SEND_CMER = "send_cmer"
    # HF-side incoming
    RECV_BRSF_OK = "recv_brsf_ok"
    RECV_BAC_OK = "recv_bac_ok"
    RECV_CIND_TEST_OK = "recv_cind_test_ok"
    RECV_CIND_GET_OK = "recv_cind_get_ok"
    RECV_CMER_OK = "recv_cmer_ok"
    # AG-side incoming
    RECV_BRSF = "recv_brsf"
    RECV_BAC = "recv_bac"
    RECV_CIND_TEST = "recv_cind_test"
    RECV_CIND_GET = "recv_cind_get"
    RECV_CMER = "recv_cmer"
    # AG-side outgoing
    SEND_BRSF_OK = "send_brsf_ok"
    SEND_BAC_OK = "send_bac_ok"
    SEND_CIND_TEST_OK = "send_cind_test_ok"
    SEND_CIND_GET_OK = "send_cind_get_ok"
    SEND_CMER_OK = "send_cmer_ok"
    # Lifecycle
    RFCOMM_DISCONNECT = "rfcomm_disconnect"


def build_hf_slc_machine() -> StateMachine[SLCState, SLCEvent]:
    sm = StateMachine(name="hfp_hf_slc", initial=SLCState.INITIAL)
    # BRSF phase (commands sent and acknowledged)
    sm.add_transition(SLCState.INITIAL, SLCEvent.SEND_BRSF, SLCState.INITIAL)
    sm.add_transition(SLCState.INITIAL, SLCEvent.RECV_BRSF_OK, SLCState.BRSF_DONE)
    # BAC phase
    sm.add_transition(SLCState.BRSF_DONE, SLCEvent.SEND_BAC, SLCState.BRSF_DONE)
    sm.add_transition(SLCState.BRSF_DONE, SLCEvent.RECV_BAC_OK, SLCState.BAC_DONE)
    # CIND test + get phase (compressed: both queries finish → CIND_DONE)
    sm.add_transition(SLCState.BAC_DONE, SLCEvent.SEND_CIND_TEST, SLCState.BAC_DONE)
    sm.add_transition(SLCState.BAC_DONE, SLCEvent.RECV_CIND_TEST_OK, SLCState.BAC_DONE)
    sm.add_transition(SLCState.BAC_DONE, SLCEvent.SEND_CIND_GET, SLCState.BAC_DONE)
    sm.add_transition(SLCState.BAC_DONE, SLCEvent.RECV_CIND_GET_OK, SLCState.CIND_DONE)
    # CMER phase
    sm.add_transition(SLCState.CIND_DONE, SLCEvent.SEND_CMER, SLCState.CIND_DONE)
    sm.add_transition(SLCState.CIND_DONE, SLCEvent.RECV_CMER_OK, SLCState.CONNECTED)
    # Disconnect from anywhere
    for s in (SLCState.INITIAL, SLCState.BRSF_DONE, SLCState.BAC_DONE,
              SLCState.CIND_DONE, SLCState.CONNECTED):
        sm.add_transition(s, SLCEvent.RFCOMM_DISCONNECT, SLCState.DISCONNECTED)
    return sm


def build_ag_slc_machine() -> StateMachine[SLCState, SLCEvent]:
    sm = StateMachine(name="hfp_ag_slc", initial=SLCState.INITIAL)
    sm.add_transition(SLCState.INITIAL, SLCEvent.RECV_BRSF, SLCState.INITIAL)
    sm.add_transition(SLCState.INITIAL, SLCEvent.SEND_BRSF_OK, SLCState.BRSF_DONE)
    sm.add_transition(SLCState.BRSF_DONE, SLCEvent.RECV_BAC, SLCState.BRSF_DONE)
    sm.add_transition(SLCState.BRSF_DONE, SLCEvent.SEND_BAC_OK, SLCState.BAC_DONE)
    sm.add_transition(SLCState.BAC_DONE, SLCEvent.RECV_CIND_TEST, SLCState.BAC_DONE)
    sm.add_transition(SLCState.BAC_DONE, SLCEvent.SEND_CIND_TEST_OK, SLCState.BAC_DONE)
    sm.add_transition(SLCState.BAC_DONE, SLCEvent.RECV_CIND_GET, SLCState.BAC_DONE)
    sm.add_transition(SLCState.BAC_DONE, SLCEvent.SEND_CIND_GET_OK, SLCState.CIND_DONE)
    sm.add_transition(SLCState.CIND_DONE, SLCEvent.RECV_CMER, SLCState.CIND_DONE)
    sm.add_transition(SLCState.CIND_DONE, SLCEvent.SEND_CMER_OK, SLCState.CONNECTED)
    for s in (SLCState.INITIAL, SLCState.BRSF_DONE, SLCState.BAC_DONE,
```

```python
class CallState(Enum):
    IDLE = "idle"
    INCOMING = "incoming"
    OUTGOING = "outgoing"
    ACTIVE = "active"
    HELD = "held"


class CallEvent(Enum):
    INCOMING_RINGING = "incoming_ringing"
    OUTGOING_DIAL = "outgoing_dial"
    ACCEPT = "accept"                # ATA
    REJECT = "reject"                # AT+CHUP from IDLE/INCOMING
    HANGUP = "hangup"                # AT+CHUP from ACTIVE/HELD
    REMOTE_ANSWERED = "remote_answered"   # AG reports outgoing answered
    HOLD = "hold"
    RESUME = "resume"


def build_call_state_machine() -> StateMachine[CallState, CallEvent]:
    sm = StateMachine(name="hfp_call", initial=CallState.IDLE)
    sm.add_transition(CallState.IDLE, CallEvent.INCOMING_RINGING, CallState.INCOMING)
    sm.add_transition(CallState.IDLE, CallEvent.OUTGOING_DIAL, CallState.OUTGOING)
    sm.add_transition(CallState.INCOMING, CallEvent.ACCEPT, CallState.ACTIVE)
    sm.add_transition(CallState.INCOMING, CallEvent.REJECT, CallState.IDLE)
    sm.add_transition(CallState.OUTGOING, CallEvent.REMOTE_ANSWERED, CallState.ACTIVE)
    sm.add_transition(CallState.OUTGOING, CallEvent.HANGUP, CallState.IDLE)
    sm.add_transition(CallState.ACTIVE, CallEvent.HOLD, CallState.HELD)
    sm.add_transition(CallState.ACTIVE, CallEvent.HANGUP, CallState.IDLE)
    sm.add_transition(CallState.HELD, CallEvent.RESUME, CallState.ACTIVE)
    sm.add_transition(CallState.HELD, CallEvent.HANGUP, CallState.IDLE)
    return sm
```

```python
from typing import NamedTuple


class IndicatorSpec(NamedTuple):
    name: str
    min_value: int
    max_value: int


DEFAULT_INDICATORS: list[IndicatorSpec] = [
    IndicatorSpec("service", 0, 1),
    IndicatorSpec("call", 0, 1),
    IndicatorSpec("callsetup", 0, 3),
    IndicatorSpec("callheld", 0, 2),
    IndicatorSpec("signal", 0, 5),
    IndicatorSpec("roam", 0, 1),
    IndicatorSpec("battchg", 0, 5),
]


@dataclass
class Indicators:
    service: int = 0
    call: int = 0
    callsetup: int = 0
    callheld: int = 0
    signal: int = 0
    roam: int = 0
    battchg: int = 0

    def to_values_list(self, specs: list[IndicatorSpec]) -> list[int]:
        return [getattr(self, s.name) for s in specs]


def format_cind_test_response(specs: list[IndicatorSpec]) -> bytes:
    """AG's reply to AT+CIND=? as on-wire bytes."""
    parts = [f'("{s.name}",({s.min_value}-{s.max_value}))' for s in specs]
    return f"\r\n+CIND: {','.join(parts)}\r\n".encode("ascii")


def format_cind_get_response(inds: Indicators,
                              specs: list[IndicatorSpec] = DEFAULT_INDICATORS) -> bytes:
    values = inds.to_values_list(specs)
    return f"\r\n+CIND: {','.join(str(v) for v in values)}\r\n".encode("ascii")


def format_ciev_event(*, indicator_name: str, value: int,
                       indicators: list[IndicatorSpec] = DEFAULT_INDICATORS) -> bytes:
    """Unsolicited +CIEV: <index>,<value> event (1-indexed)."""
    for i, spec in enumerate(indicators, start=1):
        if spec.name == indicator_name:
            return f"\r\n+CIEV: {i},{value}\r\n".encode("ascii")
    raise ValueError(f"unknown indicator name {indicator_name!r}")


def parse_cind_test_response(raw: bytes) -> list[IndicatorSpec]:
    import re
    text = raw.decode("ascii").strip()
    # Strip "+CIND: " prefix.
    if text.startswith("+CIND:"):
        text = text[len("+CIND:"):].strip()
    # Match repeated ("name",(min-max))
    specs = []
    for m in re.finditer(r'\("([^"]+)",\((\d+)-(\d+)\)\)', text):
        specs.append(IndicatorSpec(m.group(1), int(m.group(2)), int(m.group(3))))
    return specs


def parse_cind_get_response(raw: bytes, specs: list[IndicatorSpec]) -> Indicators:
    text = raw.decode("ascii").strip()
    if text.startswith("+CIND:"):
        text = text[len("+CIND:"):].strip()
    values = [int(v.strip()) for v in text.split(",")]
    if len(values) != len(specs):
        raise ValueError(
            f"+CIND get: got {len(values)} values, expected {len(specs)} "
            f"(per AT+CIND=? response)"
        )
    inds = Indicators()
    for spec, val in zip(specs, values):
        setattr(inds, spec.name, val)
    return inds
```

```python
from pybluehost.hci.constants import (
    EVENT_SYNCHRONOUS_CONNECTION_COMPLETE,
    HCI_COMMAND_PACKET, HCI_EVENT_PACKET,
    OPCODE_SETUP_SYNCHRONOUS_CONNECTION,
)


@dataclass(frozen=True)
class HCISetupSynchronousConnectionCommand:
    connection_handle: int            # u16 — the ACL handle to upgrade to SCO
    transmit_bandwidth: int           # u32
    receive_bandwidth: int            # u32
    max_latency: int                  # u16
    voice_setting: int                # u16
    retransmission_effort: int        # u8
    packet_type: int                  # u16

    @property
    def opcode(self) -> int:
        return OPCODE_SETUP_SYNCHRONOUS_CONNECTION

    def to_bytes(self) -> bytes:
        params = _struct.pack(
            "<HIIHHBH",
            self.connection_handle & 0x0FFF,
            self.transmit_bandwidth, self.receive_bandwidth,
            self.max_latency, self.voice_setting,
            self.retransmission_effort & 0xFF,
            self.packet_type,
        )
        return _struct.pack(
            "<BHB", HCI_COMMAND_PACKET, self.opcode, len(params),
        ) + params


@dataclass(frozen=True)
class HCISynchronousConnectionCompleteEvent:
    status: int
    connection_handle: int            # u16 — new SCO handle
    bd_addr: bytes                    # 6 bytes
    link_type: int                    # u8 (0=SCO, 2=eSCO)
    transmission_interval: int
    retransmission_window: int
    rx_packet_length: int             # u16
    tx_packet_length: int             # u16
    air_mode: int                     # u8

    EVENT_CODE = EVENT_SYNCHRONOUS_CONNECTION_COMPLETE

    def to_bytes(self) -> bytes:
        body = (
            _struct.pack("<BH", self.status, self.connection_handle & 0x0FFF)
            + bytes(reversed(self.bd_addr))     # HCI byte order is reversed BD_ADDR
            + _struct.pack(
                "<BBBHHB",
                self.link_type,
                self.transmission_interval,
                self.retransmission_window,
                self.rx_packet_length,
                self.tx_packet_length,
                self.air_mode,
            )
        )
        return _struct.pack(
            "<BBB", HCI_EVENT_PACKET, self.EVENT_CODE, len(body),
        ) + body

    @classmethod
    def from_bytes(cls, body: bytes) -> "HCISynchronousConnectionCompleteEvent":
        if len(body) != 17:
            raise ValueError(f"Synchronous_Connection_Complete body must be 17 bytes, got {len(body)}")
        status, conn_handle = _struct.unpack_from("<BH", body)
        bd_addr = bytes(reversed(body[3:9]))    # reverse back to standard order
        link_type, tx_int, rt_win, rx_len, tx_len, air_mode = _struct.unpack_from(
            "<BBBHHB", body, 9,
        )
        return cls(
            status=status, connection_handle=conn_handle & 0x0FFF,
            bd_addr=bd_addr, link_type=link_type,
            transmission_interval=tx_int, retransmission_window=rt_win,
            rx_packet_length=rx_len, tx_packet_length=tx_len,
            air_mode=air_mode,
        )
```

```python
from pybluehost.hci.packets import HCISCOData

logger = logging.getLogger(__name__)


_SCO_FRAME_BYTES = 60                    # HV3 / EV3 SCO payload
_CVSD_PCM_PER_FRAME = _SCO_FRAME_BYTES * 8 * 2     # 1 bit/sample, 16-bit PCM, =960 bytes
# CVSD: 60 bytes of encoded data = 480 samples = 960 bytes of int16 LE PCM.

_MSBC_PCM_PER_FRAME = 240                # 15 blocks × 8 subbands × 1 ch × 2 bytes = 240 bytes


class WavToScoOutboundWorker:
    """Reads a WAV file, encodes to the chosen HFP codec, emits HCISCOData packets.

    Plan A.4 ships file loopback only; no realtime pacing (the test bus delivers
    packets as fast as `send_packet` accepts them). Real-time pacing belongs
    to Plan A.6 (sounddevice integration).
    """

    def __init__(
        self, *, wav_path: str, sco_handle: int, codec: str,
        send_packet: Callable[[HCISCOData], Awaitable[None]],
    ) -> None:
        if codec not in ("CVSD", "mSBC"):
            raise ValueError(f"unsupported HFP codec: {codec!r}")
        self._wav_path = wav_path
        self._sco_handle = sco_handle
        self._codec = codec
        self._send = send_packet

    async def run_to_completion(self) -> None:
        with wave.open(self._wav_path, "rb") as w:
            sample_rate = w.getframerate()
            channels = w.getnchannels()
            sample_width = w.getsampwidth()
            if sample_width != 2:
                raise ValueError(
                    f"SCO outbound: WAV must be 16-bit PCM (got {sample_width * 8}-bit)"
                )
            if channels != 1:
                raise ValueError(f"SCO outbound: WAV must be mono (got {channels} channels)")
            if self._codec == "CVSD" and sample_rate != 8000:
                raise ValueError(
                    f"CVSD requires 8 kHz sample rate, got {sample_rate}"
                )
            if self._codec == "mSBC" and sample_rate != 16000:
                raise ValueError(
                    f"mSBC requires 16 kHz sample rate, got {sample_rate}"
                )
            pcm_per_frame = _CVSD_PCM_PER_FRAME if self._codec == "CVSD" else _MSBC_PCM_PER_FRAME
            encoder = CVSDEncoder() if self._codec == "CVSD" else MSBCEncoder()
            while True:
                pcm_chunk = w.readframes(pcm_per_frame // 2)
                if len(pcm_chunk) < pcm_per_frame:
                    break       # silently drop trailing partial frame
                encoded = encoder.encode(pcm_chunk)
                if len(encoded) != _SCO_FRAME_BYTES:
                    raise RuntimeError(
                        f"encoder produced {len(encoded)} bytes; expected {_SCO_FRAME_BYTES} "
                        f"(check codec config — Plan A.1 SBC bitpool 26 / blocks 15 yields 57+3 wrapper)"
                    )
                pkt = HCISCOData(
                    handle=self._sco_handle, packet_status=0, data=encoded,
                )
```

```python
class ScoToWavInboundWorker:
    """Receives HCISCOData packets, decodes via CVSDDecoder/MSBCDecoder, appends to WAV."""

    def __init__(
        self, *, wav_path: str, sample_rate: int, channels: int = 1, codec: str,
    ) -> None:
        if codec not in ("CVSD", "mSBC"):
            raise ValueError(f"unsupported HFP codec: {codec!r}")
        self._codec = codec
        self._decoder = CVSDDecoder() if codec == "CVSD" else MSBCDecoder()
        self._wav = wave.open(wav_path, "wb")
        self._wav.setnchannels(channels)
        self._wav.setsampwidth(2)
        self._wav.setframerate(sample_rate)

    async def feed(self, pkt: HCISCOData) -> None:
        encoded = pkt.data
        if self._codec == "mSBC":
            # Strip 2-byte H2 header + 1-byte pad.
            if len(encoded) != 60 or encoded[0] != 0x01:
                logger.warning(
                    "ScoToWavInboundWorker: malformed mSBC envelope, dropping packet"
                )
                return
            encoded = encoded[2:-1]    # 57-byte SBC frame
        pcm = self._decoder.decode(encoded)
        self._wav.writeframes(pcm)

    async def close(self) -> None:
        if self._wav is not None:
            self._wav.close()
            self._wav = None
```

```python
class HFPClient:
    """HFP HF (Hands-Free) role — drives SLC + call state on top of an RFCOMM channel."""

    PROFILE_NAME = "HFPClient"
    SDP_RECORD = "hfp_hf.yaml"

    def __init__(self, rfcomm_channel, config: HFPConfig | None = None) -> None:
        self._ch = rfcomm_channel
        self._config = config or HFPConfig.default_hf()
        self._slc_sm = build_hf_slc_machine()
        self._buffer = bytearray()
        self._pending_result: asyncio.Future[bool] | None = None
        self._last_responses: list[ATResponse] = []
        self._ag_brsf: int | None = None
        self._ag_indicators: list[IndicatorSpec] | None = None
        self._ag_indicator_values: Indicators | None = None
        rfcomm_channel.on_data(self._on_data)

    @property
    def slc_state(self) -> SLCState:
        return self._slc_sm.current_state

    async def run_slc(self, *, timeout: float = 10.0) -> None:
        """Drive SLC three-phase handshake to CONNECTED, or raise TimeoutError."""
        async def _run():
            # BRSF
            await self._exchange(
                ATCommand(name="+BRSF", args=[str(self._config.features)])
            )
            self._slc_sm.fire(SLCEvent.SEND_BRSF)
            self._slc_sm.fire(SLCEvent.RECV_BRSF_OK)
            # BAC
            await self._exchange(
                ATCommand(name="+BAC",
                           args=[str(c) for c in self._config.codecs])
            )
            self._slc_sm.fire(SLCEvent.SEND_BAC)
            self._slc_sm.fire(SLCEvent.RECV_BAC_OK)
            # CIND test
            await self._exchange(ATCommand(name="+CIND", args=["?"]))
            self._slc_sm.fire(SLCEvent.SEND_CIND_TEST)
            self._slc_sm.fire(SLCEvent.RECV_CIND_TEST_OK)
            # CIND get
            await self._exchange(ATCommand(name="+CIND", is_query=True))
            self._slc_sm.fire(SLCEvent.SEND_CIND_GET)
            self._slc_sm.fire(SLCEvent.RECV_CIND_GET_OK)
            # CMER
            await self._exchange(
                ATCommand(name="+CMER", args=["3", "0", "0", "1"])
            )
            self._slc_sm.fire(SLCEvent.SEND_CMER)
            self._slc_sm.fire(SLCEvent.RECV_CMER_OK)
        await asyncio.wait_for(_run(), timeout=timeout)

    async def _exchange(self, cmd: ATCommand) -> bool:
        """Send a command and await OK/ERROR; return ok bool."""
        self._pending_result = asyncio.get_running_loop().create_future()
        self._last_responses = []
        await self._ch.send(format_at_command(cmd))
        return await self._pending_result

    async def _on_data(self, data: bytes) -> None:
        self._buffer.extend(data)
        # Greedily extract complete CRLF-terminated chunks.
        while True:
            chunk_end = self._find_complete_at_line()
            if chunk_end is None:
                return
            line = self._buffer[:chunk_end].decode("ascii", errors="replace")
            del self._buffer[:chunk_end]
            try:
                parsed = parse_at_line(line)
            except Exception:
                logger.exception("HFPClient: bad AT line %r", line)
                continue
            await self._handle_parsed(parsed)

    def _find_complete_at_line(self) -> int | None:
        # Try to find a complete `\r\nX\r\n` or `AT...\r\n` chunk.
        if self._buffer.startswith(b"\r\n"):
            # Find closing CRLF after the leading one.
            idx = self._buffer.find(b"\r\n", 2)
            if idx == -1:
                return None
            return idx + 2
        else:
            idx = self._buffer.find(b"\r\n")
            if idx == -1:
                return None
            return idx + 2

    async def _handle_parsed(self, parsed) -> None:
        if isinstance(parsed, ATResponse):
            self._last_responses.append(parsed)
            # Remember significant responses.
            if parsed.name == "+BRSF" and parsed.args:
                self._ag_brsf = int(parsed.args[0])
            elif parsed.name == "+CIND":
                # +CIND has two flavours: test (with parens) vs get (with ints).
                if any("(" in a for a in parsed.args):
                    self._ag_indicators = [
                        IndicatorSpec(*spec) for spec in
                        # crude parse from response string; for full handling use the format helpers
                        []
                    ]
                else:
                    if self._ag_indicators:
                        # populate values
                        pass
        elif isinstance(parsed, ATResult):
            if self._pending_result is not None and not self._pending_result.done():
                self._pending_result.set_result(parsed.ok)
                self._pending_result = None
```

```python
    samples = [int(amplitude * math.sin(2 * math.pi * freq_hz * i / sample_rate))
               for i in range(n)]
    with wave.open(str(path), "wb") as w:
        w.setnchannels(1); w.setsampwidth(2); w.setframerate(sample_rate)
        w.writeframes(b"".join(struct.pack("<h", s) for s in samples))


async def test_hfp_full_slc_plus_sco_loopback_cvsd(tmp_path):
    hf_stack = await Stack.virtual()
    ag_stack = await Stack.virtual()
    src_wav = tmp_path / "src_8k.wav"
    dst_wav = tmp_path / "dst_8k.wav"
    _synth_mono_wav(src_wav, sample_rate=8000)
    try:
        # ACL connect
        await hf_stack.gap.connect(ag_stack.local_address)
        acl_handle = next(iter(hf_stack.l2cap._classic_handles))

        # AG side: register HFPServer on its RFCOMM channel 13 (per yaml).
        async def _accept_hfp_channel(channel):
            HFPServer(rfcomm_channel=channel, config=HFPConfig.default_ag())
        await ag_stack.rfcomm.listen(server_channel=13, handler=_accept_hfp_channel)

        # HF side: RFCOMM connect to AG channel 13, run SLC.
        hf_channel = await hf_stack.rfcomm.connect(acl_handle=acl_handle, server_channel=13)
        client = HFPClient(rfcomm_channel=hf_channel, config=HFPConfig.default_hf())
        await client.run_slc(timeout=5.0)
        assert client.slc_state == SLCState.CONNECTED

        # SCO link setup (CVSD)
        sco_event = await hf_stack.hci.setup_sco(
            connection_handle=acl_handle, voice_setting=0x0060,
        )
        assert sco_event.status == 0x00
        sco_handle = sco_event.connection_handle

        # Wire inbound + outbound workers on AG side; HF will push outbound.
        ag_inbound = ScoToWavInboundWorker(
            wav_path=str(dst_wav), sample_rate=8000, channels=1, codec="CVSD",
        )
        ag_stack.hci._on_sco_data = lambda pkt: asyncio.create_task(ag_inbound.feed(pkt))

        # HF outbound worker streams the WAV via the HCI transport.
        async def send_pkt(pkt):
            await hf_stack.hci._transport.send(pkt.to_bytes())
        outbound = WavToScoOutboundWorker(
            wav_path=str(src_wav), sco_handle=sco_handle,
            codec="CVSD", send_packet=send_pkt,
        )
        await outbound.run_to_completion()
        await asyncio.sleep(0.05)
        await ag_inbound.close()

        # PSNR sanity on round-trip
        with wave.open(str(src_wav), "rb") as r1, wave.open(str(dst_wav), "rb") as r2:
            n = min(r1.getnframes(), r2.getnframes()) - 200    # codec warmup
            r1.readframes(200); r2.readframes(200)
            src = struct.unpack(f"<{n}h", r1.readframes(n))
            dst = struct.unpack(f"<{n}h", r2.readframes(n))
        err = [a - b for a, b in zip(src, dst)]
        mse = sum(e * e for e in err) / max(len(err), 1)
        psnr = 10 * math.log10((32767 ** 2) / max(mse, 1))
        assert psnr > 25, f"CVSD round-trip PSNR {psnr:.2f} dB below 25 dB"

    finally:
        await hf_stack.close()
        await ag_stack.close()
```
