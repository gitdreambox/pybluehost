> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add Bluetooth remote-control on top of A2DP: ship `AVRCPController` + `AVRCPTarget` profile classes and a `pybluehost app avrcp-control --target=<addr> --cmd=play` CLI that drives a real phone's playback.

**Architecture:** AVCTP is the transport (L2CAP PSM 0x0017, one signaling channel per peer) — much simpler than AVDTP: just message framing with fragmentation. AVRCP is the application layer riding on AVCTP, carrying two flavours of AV/C messages: **PASS_THROUGH** (single-button commands like PLAY/PAUSE/VOLUME_UP, AV/C §9.4) and **VENDOR_DEPENDENT** with BT-SIG company ID (AVRCP-specific metadata + notifications, AVRCP v1.6 §6.7). Profile registration reuses the `Stack.register_profile` surface added in Plan A.2 Task 14; the `VirtualClassicLink` dynamic-PSM bridge from A.2 Task 15 already supports 0x0017 — Plan A.3 just adds it to the allow-list.

**Tech Stack:** Python 3.10+, asyncio, dataclasses, struct, pytest with `asyncio_mode=auto`. No new third-party deps. Hard dependency on Plan A.2 (Stack.register_profile, VirtualClassicLink dynamic-PSM bridge, profile-registration pattern).

Per AVCTP v1.4 §6.1, the SINGLE-packet header is **3 bytes**:

```
Byte 0:  Transaction Label (4 bits) | Packet Type (2 bits) | C/R (1 bit) | IPID (1 bit)
Bytes 1-2: PID (BE u16)
Bytes 3..: payload
```

For START-fragmented packets the header gains a Number-of-Packets byte after the 1st byte; CONTINUE/END have a 1-byte header. Task 3 adds fragmentation.

```python
"""AVCTP message frame encode/decode (AVCTP v1.4 §6)."""
from __future__ import annotations

from dataclasses import dataclass

from pybluehost.avctp import constants as c


@dataclass(frozen=True)
class AVCTPMessage:
    """A single (non-fragmented) AVCTP message.

    Fragmentation/reassembly is handled at the session layer (Task 3+4).
    """

    transaction_label: int      # 0..15
    cr: int                     # 0 = command, 1 = response
    ipid: bool                  # set when peer can't handle PID
    profile_id: int             # 16-bit
    payload: bytes              # AV/C frame bytes

    def __post_init__(self) -> None:
        if not 0 <= self.transaction_label <= c.MAX_TRANSACTION_LABEL:
            raise ValueError(f"transaction_label {self.transaction_label} out of 0..15")
        if self.cr not in (c.CR_COMMAND, c.CR_RESPONSE):
            raise ValueError(f"cr must be 0 or 1, got {self.cr}")
        if not 0 <= self.profile_id <= 0xFFFF:
            raise ValueError(f"profile_id 0x{self.profile_id:X} out of u16 range")

    def to_bytes(self) -> bytes:
        header_byte = (
            (self.transaction_label << 4)
            | (c.PACKET_TYPE_SINGLE << 2)
            | (self.cr << 1)
            | (1 if self.ipid else 0)
        )
        return bytes([header_byte]) + self.profile_id.to_bytes(2, "big") + self.payload

    @classmethod
    def from_bytes(cls, data: bytes) -> "AVCTPMessage":
        if len(data) < 3:
            raise ValueError(f"AVCTP frame too short ({len(data)} bytes; need ≥ 3)")
        h = data[0]
        packet_type = (h >> 2) & 0x03
        if packet_type != c.PACKET_TYPE_SINGLE:
            raise ValueError(
                f"AVCTP: expected SINGLE packet but got fragment (packet_type={packet_type})"
            )
        transaction_label = (h >> 4) & 0x0F
        cr = (h >> 1) & 0x01
        ipid = bool(h & 0x01)
        profile_id = int.from_bytes(data[1:3], "big")
        return cls(
            transaction_label=transaction_label, cr=cr, ipid=ipid,
            profile_id=profile_id, payload=bytes(data[3:]),
        )
```

## Task 3: AVCTP fragmentation (START / CONTINUE / END)

Per AVCTP v1.4 §6.2, fragmentation kicks in when a message exceeds the L2CAP MTU. Header layouts:

```
START header (3 bytes):
  Byte 0: trans_label(4) | packet_type(2)=01 | cr(1) | ipid(1)
  Byte 1: Number of Packets (NoP)
  Bytes 2-3: PID (BE u16)
  Bytes 4..: chunk of payload

CONTINUE/END header (1 byte):
  Byte 0: trans_label(4) | packet_type(2)=10|11 | cr(1) | ipid(1)
  Bytes 1..: chunk of payload
```

The receiver concatenates chunks until END arrives.

```python
def fragment_message(msg: AVCTPMessage, *, mtu: int) -> list[bytes]:
    """Split an AVCTPMessage into one or more L2CAP frames per AVCTP v1.4 §6.2.

    Returns a list of byte strings, each ≤ mtu. If the entire message fits in
    one frame, returns `[msg.to_bytes()]`.
    """
    if mtu < 4:
        raise ValueError(f"mtu {mtu} too small (need ≥ 4 for any AVCTP frame)")
    single = msg.to_bytes()
    if len(single) <= mtu:
        return [single]

    # Fragmentation needed.
    # START frame header = 4 bytes (1 + NoP + PID).
    # CONTINUE/END header = 1 byte.
    payload = msg.payload
    pid = msg.profile_id
    label = msg.transaction_label
    cr = msg.cr
    ipid = msg.ipid

    start_payload_cap = mtu - 4
    cont_payload_cap = mtu - 1

    # First, figure out how many frames we'll need so we can fill the NoP byte.
    consumed = start_payload_cap
    n_frames = 1
    while consumed < len(payload):
        consumed += cont_payload_cap
        n_frames += 1
    if n_frames > 0xFF:
        raise ValueError(f"message requires {n_frames} fragments — exceeds NoP byte (255)")

    frames: list[bytes] = []
    # START
    header = (label << 4) | (c.PACKET_TYPE_START << 2) | (cr << 1) | (1 if ipid else 0)
    start_chunk = payload[:start_payload_cap]
    frames.append(bytes([header, n_frames]) + pid.to_bytes(2, "big") + start_chunk)
    offset = start_payload_cap

    while offset < len(payload):
        remaining = len(payload) - offset
        chunk = payload[offset:offset + cont_payload_cap]
        is_end = remaining <= cont_payload_cap
        ptype = c.PACKET_TYPE_END if is_end else c.PACKET_TYPE_CONTINUE
        header = (label << 4) | (ptype << 2) | (cr << 1) | (1 if ipid else 0)
        frames.append(bytes([header]) + chunk)
        offset += cont_payload_cap

    return frames


class Reassembler:
    """Stateful AVCTP fragment reassembler — feed frames in order, get a message back."""

    def __init__(self) -> None:
        self._buf: bytearray = bytearray()
        self._label: int | None = None
        self._cr: int | None = None
        self._ipid: bool | None = None
        self._pid: int | None = None
        self._expected_packets: int = 0
        self._received_packets: int = 0

    def feed(self, frame: bytes) -> AVCTPMessage | None:
        if len(frame) < 1:
            raise ValueError("AVCTP fragment empty")
        h = frame[0]
        packet_type = (h >> 2) & 0x03
        label = (h >> 4) & 0x0F
        cr = (h >> 1) & 0x01
        ipid = bool(h & 0x01)

        if packet_type == c.PACKET_TYPE_SINGLE:
            return AVCTPMessage.from_bytes(frame)

        if packet_type == c.PACKET_TYPE_START:
            if len(frame) < 4:
                raise ValueError("AVCTP START frame too short")
            self._buf = bytearray(frame[4:])
            self._label = label
            self._cr = cr
            self._ipid = ipid
            self._pid = int.from_bytes(frame[2:4], "big")
            self._expected_packets = frame[1]
            self._received_packets = 1
            return None

        if packet_type in (c.PACKET_TYPE_CONTINUE, c.PACKET_TYPE_END):
            if self._label is None:
                raise ValueError("unexpected CONTINUE/END without START")
            if label != self._label:
                raise ValueError(
                    f"transaction label mismatch: got {label} expected {self._label}"
                )
            self._buf.extend(frame[1:])
            self._received_packets += 1
            if packet_type == c.PACKET_TYPE_END:
                msg = AVCTPMessage(
                    transaction_label=self._label, cr=self._cr, ipid=self._ipid,
                    profile_id=self._pid, payload=bytes(self._buf),
                )
                self._reset()
                return msg
            return None

        raise ValueError(f"unknown AVCTP packet_type {packet_type}")

    def _reset(self) -> None:
        self._buf = bytearray()
        self._label = None
        self._cr = None
        self._ipid = None
        self._pid = None
        self._expected_packets = 0
        self._received_packets = 0
```

```python
"""AVCTPSession — owns one L2CAP channel, routes by transaction label."""
from __future__ import annotations

import asyncio
import logging
from typing import Any, Awaitable, Callable

from pybluehost.avctp import constants as c
from pybluehost.avctp.messages import AVCTPMessage, Reassembler, fragment_message

logger = logging.getLogger(__name__)


class AVCTPSession:
    def __init__(self, channel: Any) -> None:
        self._channel = channel
        self._reassembler = Reassembler()
        self._next_label = 0
        self._pending: dict[int, asyncio.Future[AVCTPMessage]] = {}
        self._message_handler: Callable[[AVCTPMessage], Awaitable[None]] | None = None
        channel.on_data(self._on_channel_data)

    @property
    def mtu(self) -> int:
        return getattr(self._channel, "mtu", 672)

    def on_message(self, cb: Callable[[AVCTPMessage], Awaitable[None]]) -> None:
        """Register the callback invoked when a command arrives."""
        self._message_handler = cb

    async def send(self, msg: AVCTPMessage) -> None:
        """Send a pre-built AVCTPMessage; fragments per current MTU."""
        for frame in fragment_message(msg, mtu=self.mtu):
            await self._channel.send(frame)

    async def send_command(
        self, *, profile_id: int, payload: bytes, timeout: float = 5.0,
    ) -> AVCTPMessage:
        """Send a command and await the matching response."""
        label = self._next_label
        self._next_label = (self._next_label + 1) & 0x0F
        future: asyncio.Future[AVCTPMessage] = asyncio.get_running_loop().create_future()
        self._pending[label] = future
        msg = AVCTPMessage(
            transaction_label=label, cr=c.CR_COMMAND, ipid=False,
            profile_id=profile_id, payload=payload,
        )
        await self.send(msg)
        try:
            return await asyncio.wait_for(future, timeout=timeout)
        finally:
            self._pending.pop(label, None)

    async def _on_channel_data(self, data: bytes) -> None:
        try:
            msg = self._reassembler.feed(data)
        except ValueError:
            logger.exception("AVCTP reassembly error")
            return
        if msg is None:
            return    # waiting for more fragments
        if msg.cr == c.CR_RESPONSE:
            fut = self._pending.get(msg.transaction_label)
            if fut is not None and not fut.done():
                fut.set_result(msg)
            # else: unsolicited response — silently drop.
            return
        # COMMAND → user handler.
        if self._message_handler is not None:
            await self._message_handler(msg)
        else:
            logger.warning(
                "AVCTP: incoming command label=%d pid=0x%04X has no handler",
                msg.transaction_label, msg.profile_id,
            )
```

Per AV/C Digital Interface General Spec §5 (frame layout) + AVRCP v1.6.2 §4.4 (PASS_THROUGH):

```
Byte 0: ctype (4 bits) | subunit_type (5 bits) | subunit_id (3 bits)   ─ split across two bytes
        Actually:
Byte 0: ctype (4 bits) | reserved (4 bits, =0)
Byte 1: subunit_type (5 bits) | subunit_id (3 bits)
Byte 2: opcode (8 bits)              — 0x7C = PASS_THROUGH
Byte 3: operation_id (7 bits) | state (1 bit)  ─ state=0 press, 1 release
Byte 4: operation_data_length        — usually 0 for PASS_THROUGH
Bytes 5..: operation_data            — usually empty
```

CTYPE values (AV/C §7.3):
- 0x0 CONTROL (PASS_THROUGH command)
- 0x8 NOT_IMPLEMENTED (response when peer doesn't support)
- 0x9 ACCEPTED (response success)
- 0xA REJECTED (response failure)
- 0xC STABLE (response for STATUS commands)
- 0xD CHANGED (notification response)
- 0xF INTERIM (notification interim response)

Subunit type for AVRCP = 0x09 (PANEL), subunit_id = 0x00 (single instance).

PASS_THROUGH operation IDs (AV/C TA 1999008 Table 9.21):
- 0x44 PLAY
- 0x45 STOP
- 0x46 PAUSE
- 0x47 RECORD
- 0x48 REWIND
- 0x49 FAST_FORWARD
- 0x4B FORWARD (next track)
- 0x4C BACKWARD (prev track)
- 0x41 VOLUME_UP
- 0x42 VOLUME_DOWN
- 0x43 MUTE

```python
"""AVRCP v1.6.2 Controller + Target — Plan A.3 (PASS_THROUGH + notifications)."""
from __future__ import annotations

from dataclasses import dataclass
from enum import IntEnum


# AV/C Digital Interface General Spec §7.3: ctype values.
class AvcCType(IntEnum):
    CONTROL = 0x0
    STATUS = 0x1
    SPECIFIC_INQUIRY = 0x2
    NOTIFY = 0x3
    GENERAL_INQUIRY = 0x4
    NOT_IMPLEMENTED = 0x8
    ACCEPTED = 0x9
    REJECTED = 0xA
    IN_TRANSITION = 0xB
    STABLE = 0xC
    CHANGED = 0xD
    INTERIM = 0xF


AVRCP_SUBUNIT_PANEL = 0x09
AVRCP_SUBUNIT_UNIT = 0x1F           # used for UNIT_INFO commands

# AV/C opcodes (AV/C TA 1999008 Table 7.4).
AVRCP_PASS_THROUGH_OPCODE = 0x7C
AVRCP_UNIT_INFO_OPCODE = 0x30
AVRCP_VENDOR_DEPENDENT_OPCODE = 0x00

# PASS_THROUGH operation IDs (AV/C Panel Subunit §6, Table 9.21).
AVRCP_OPERATION_VOLUME_UP = 0x41
AVRCP_OPERATION_VOLUME_DOWN = 0x42
AVRCP_OPERATION_MUTE = 0x43
AVRCP_OPERATION_PLAY = 0x44
AVRCP_OPERATION_STOP = 0x45
AVRCP_OPERATION_PAUSE = 0x46
AVRCP_OPERATION_RECORD = 0x47
AVRCP_OPERATION_REWIND = 0x48
AVRCP_OPERATION_FAST_FORWARD = 0x49
AVRCP_OPERATION_FORWARD = 0x4B       # next track
AVRCP_OPERATION_BACKWARD = 0x4C      # previous track


@dataclass(frozen=True)
class AvcFrame:
    """AV/C frame — used for both PASS_THROUGH and VENDOR_DEPENDENT."""
    ctype: AvcCType
    subunit_type: int
    subunit_id: int
    opcode: int
    operands: bytes

    def to_bytes(self) -> bytes:
        b0 = (int(self.ctype) & 0x0F)        # high 4 bits reserved (0)
        b1 = ((self.subunit_type & 0x1F) << 3) | (self.subunit_id & 0x07)
        return bytes([b0, b1, self.opcode]) + self.operands

    @classmethod
    def from_bytes(cls, data: bytes) -> "AvcFrame":
        if len(data) < 3:
            raise ValueError(f"AV/C frame too short ({len(data)} bytes)")
        ctype = AvcCType(data[0] & 0x0F)
        subunit_type = (data[1] >> 3) & 0x1F
        subunit_id = data[1] & 0x07
        opcode = data[2]
        return cls(ctype=ctype, subunit_type=subunit_type, subunit_id=subunit_id,
                   opcode=opcode, operands=bytes(data[3:]))


@dataclass(frozen=True)
class PassThroughFrame:
    """Decoded PASS_THROUGH AV/C frame."""
    ctype: AvcCType
    subunit_type: int
    subunit_id: int
    opcode: int                    # always 0x7C
    operation_id: int              # 7-bit
    state_press: bool              # True = press (state bit=0), False = release (state bit=1)
    operation_data: bytes          # usually empty


def encode_pass_through_command(
    *, operation_id: int, state_press: bool = True,
    operation_data: bytes = b"",
) -> bytes:
    """Encode a PASS_THROUGH **command** (ctype=CONTROL) AV/C frame body."""
    if not 0 <= operation_id <= 0x7F:
        raise ValueError(f"operation_id 0x{operation_id:02X} out of 7-bit range")
    op_state = (operation_id & 0x7F) | (0x00 if state_press else 0x80)
    operands = bytes([op_state, len(operation_data)]) + operation_data
    return AvcFrame(
        ctype=AvcCType.CONTROL, subunit_type=AVRCP_SUBUNIT_PANEL, subunit_id=0,
        opcode=AVRCP_PASS_THROUGH_OPCODE, operands=operands,
    ).to_bytes()


def encode_pass_through_response(
    *, ctype: AvcCType, operation_id: int, state_press: bool = True,
    operation_data: bytes = b"",
) -> bytes:
    """Encode a PASS_THROUGH **response** with a chosen ctype (ACCEPTED, REJECTED, ...)."""
    op_state = (operation_id & 0x7F) | (0x00 if state_press else 0x80)
    operands = bytes([op_state, len(operation_data)]) + operation_data
    return AvcFrame(
        ctype=ctype, subunit_type=AVRCP_SUBUNIT_PANEL, subunit_id=0,
        opcode=AVRCP_PASS_THROUGH_OPCODE, operands=operands,
    ).to_bytes()


def decode_pass_through_frame(data: bytes) -> PassThroughFrame:
    if len(data) < 5:
        raise ValueError(f"PASS_THROUGH frame too short ({len(data)} bytes; need ≥ 5)")
    frame = AvcFrame.from_bytes(data)
    if frame.opcode != AVRCP_PASS_THROUGH_OPCODE:
        raise ValueError(
            f"not a PASS_THROUGH frame (opcode=0x{frame.opcode:02X}, "
            f"expected 0x{AVRCP_PASS_THROUGH_OPCODE:02X})"
        )
    if len(frame.operands) < 2:
        raise ValueError("PASS_THROUGH operands too short")
    op_state = frame.operands[0]
    operation_id = op_state & 0x7F
    state_press = (op_state & 0x80) == 0
    data_len = frame.operands[1]
    operation_data = frame.operands[2:2 + data_len]
    return PassThroughFrame(
        ctype=frame.ctype, subunit_type=frame.subunit_type, subunit_id=frame.subunit_id,
        opcode=frame.opcode, operation_id=operation_id, state_press=state_press,
        operation_data=bytes(operation_data),
    )
```

```python
# AVRCP v1.6.2 §6.6: BT-SIG company ID carried in VENDOR_DEPENDENT frames.
AVRCP_BT_SIG_COMPANY_ID = 0x001958

# AVRCP PDU IDs (§6.5.5).
AVRCP_PDU_GET_PLAY_STATUS = 0x30
AVRCP_PDU_REGISTER_NOTIFICATION = 0x31

# AVRCP event IDs (§6.7.2).
AVRCP_EVENT_PLAYBACK_STATUS_CHANGED = 0x01
AVRCP_EVENT_TRACK_CHANGED = 0x02
AVRCP_EVENT_VOLUME_CHANGED = 0x0D


@dataclass(frozen=True)
class VendorDependentMessage:
    ctype: AvcCType
    subunit_type: int
    subunit_id: int
    company_id: int            # 24-bit (0x001958 for AVRCP)
    pdu_id: int
    packet_type: int           # AVRCP fragmentation (within VENDOR_DEPENDENT — distinct from AVCTP)
    parameters: bytes


def encode_vendor_dependent_command(
    *, ctype: AvcCType, pdu_id: int, parameters: bytes,
    company_id: int = AVRCP_BT_SIG_COMPANY_ID,
) -> bytes:
    operands = (
        company_id.to_bytes(3, "big")
        + bytes([pdu_id, 0])    # packet_type = 0 (single)
        + len(parameters).to_bytes(2, "big")
        + parameters
    )
    return AvcFrame(
        ctype=ctype, subunit_type=AVRCP_SUBUNIT_PANEL, subunit_id=0,
        opcode=AVRCP_VENDOR_DEPENDENT_OPCODE, operands=operands,
    ).to_bytes()

    *, event_id: int, playback_interval: int = 0,
) -> bytes:

    playback_interval is the polling interval in seconds for
    PLAYBACK_POS_CHANGED; 0 for events that fire on change.

    params = bytes([event_id]) + playback_interval.to_bytes(4, "big")
    return encode_vendor_dependent_command(
        ctype=AvcCType.NOTIFY, pdu_id=AVRCP_PDU_REGISTER_NOTIFICATION, parameters=params,
    )


def encode_register_notification_response(
    *, ctype: AvcCType, event_id: int, event_params: bytes,
) -> bytes:
    """Target → controller notification response.

    `ctype` is INTERIM on subscription (carries current state) or CHANGED when
    the event fires (carries new state).
    """
    params = bytes([event_id]) + event_params
    return encode_vendor_dependent_command(
        ctype=ctype, pdu_id=AVRCP_PDU_REGISTER_NOTIFICATION, parameters=params,
    )


def decode_vendor_dependent_frame(data: bytes) -> VendorDependentMessage:
    frame = AvcFrame.from_bytes(data)
    if frame.opcode != AVRCP_VENDOR_DEPENDENT_OPCODE:
        raise ValueError(
            f"not VENDOR_DEPENDENT (opcode=0x{frame.opcode:02X})"
        )
    if len(frame.operands) < 7:
        raise ValueError(f"VENDOR_DEPENDENT operands too short ({len(frame.operands)})")
    company_id = int.from_bytes(frame.operands[0:3], "big")
    if company_id != AVRCP_BT_SIG_COMPANY_ID:
        raise ValueError(
            f"AVRCP requires BT-SIG company ID 0x{AVRCP_BT_SIG_COMPANY_ID:06X}, "
            f"got 0x{company_id:06X}"
        )
    pdu_id = frame.operands[3]
    packet_type = frame.operands[4]
    param_len = int.from_bytes(frame.operands[5:7], "big")
    if 7 + param_len > len(frame.operands):
        raise ValueError(
            f"VENDOR_DEPENDENT param_len {param_len} exceeds operand bytes"
        )
    parameters = bytes(frame.operands[7:7 + param_len])
    return VendorDependentMessage(
        ctype=frame.ctype, subunit_type=frame.subunit_type, subunit_id=frame.subunit_id,
        company_id=company_id, pdu_id=pdu_id, packet_type=packet_type, parameters=parameters,
    )
```

```python
import asyncio

from pybluehost.avctp.constants import AVCTP_PSM, AVRCP_PROFILE_ID


class AVRCPController:
    """Controller role: sends PASS_THROUGH commands + subscribes to notifications."""

    PROFILE_NAME = "AVRCPController"
    PSM = AVCTP_PSM
    SDP_RECORD = "avrcp_controller.yaml"

    def __init__(self, session, stack=None) -> None:
        self._session = session
        self._stack = stack
        self._notification_handlers: dict[int, list] = {}     # event_id → [callbacks]

    async def press(
        self, operation_id: int, *, hold_seconds: float = 0.05, timeout: float = 5.0,
    ) -> None:
        """Send a PASS_THROUGH press followed by a release. Most targets reject
        a press without a matching release inside ~2 seconds."""
        await self._send_pass_through(operation_id, state_press=True, timeout=timeout)
        await asyncio.sleep(hold_seconds)
        await self._send_pass_through(operation_id, state_press=False, timeout=timeout)

    async def _send_pass_through(
        self, operation_id: int, *, state_press: bool, timeout: float,
    ) -> None:
        payload = encode_pass_through_command(
            operation_id=operation_id, state_press=state_press,
        )
        reply = await self._session.send_command(
            profile_id=AVRCP_PROFILE_ID, payload=payload, timeout=timeout,
        )
        # Validate: PASS_THROUGH response with ctype ACCEPTED.
        frame = decode_pass_through_frame(reply.payload)
        if frame.ctype != AvcCType.ACCEPTED:
            raise RuntimeError(
                f"PASS_THROUGH op=0x{operation_id:02X} rejected "
                f"(ctype={frame.ctype.name})"
            )

    def on_notification(self, event_id: int, cb) -> None:
        """Register a callback invoked when `event_id` fires (after subscribe())."""
        self._notification_handlers.setdefault(event_id, []).append(cb)

    async def subscribe(self, event_id: int, *, timeout: float = 5.0) -> bytes:
        """Send REGISTER_NOTIFICATION; returns INTERIM event_params (current state)."""
        cmd = encode_register_notification_command(event_id=event_id)
        reply = await self._session.send_command(
            profile_id=AVRCP_PROFILE_ID, payload=cmd, timeout=timeout,
        )
        msg = decode_vendor_dependent_frame(reply.payload)
        if msg.ctype != AvcCType.INTERIM:
            raise RuntimeError(
                f"REGISTER_NOTIFICATION event=0x{event_id:02X} "
                f"got unexpected ctype={msg.ctype.name}"
            )
        # First byte of parameters is the event_id (echo); rest is event_params.
        return msg.parameters[1:]

    async def _on_async_notification(self, frame_payload: bytes) -> None:
        """Called when an unsolicited CHANGED notification arrives."""
        msg = decode_vendor_dependent_frame(frame_payload)
        if msg.pdu_id != AVRCP_PDU_REGISTER_NOTIFICATION:
            return
        event_id = msg.parameters[0]
        event_params = msg.parameters[1:]
        for cb in self._notification_handlers.get(event_id, []):
            res = cb(event_params)
            if asyncio.iscoroutine(res):
                await res
```

```python
_DEFAULT_OPS = frozenset({
    AVRCP_OPERATION_PLAY, AVRCP_OPERATION_PAUSE, AVRCP_OPERATION_STOP,
    AVRCP_OPERATION_FORWARD, AVRCP_OPERATION_BACKWARD,
    AVRCP_OPERATION_VOLUME_UP, AVRCP_OPERATION_VOLUME_DOWN, AVRCP_OPERATION_MUTE,
})


class AVRCPTarget:
    """Target role: receives PASS_THROUGH + emits CHANGED notifications."""

    PROFILE_NAME = "AVRCPTarget"
    PSM = AVCTP_PSM
    SDP_RECORD = "avrcp_target.yaml"

    def __init__(
        self, session, stack=None, *,
        supported_operations: set[int] | None = None,
    ) -> None:
        self._session = session
        self._stack = stack
        self._supported = supported_operations or _DEFAULT_OPS
        self._button_cb = None
        self._current_volume = 0x7F        # max per AVRCP §6.13
        self._current_play_status = 0x00   # 0=STOPPED, 1=PLAYING, 2=PAUSED, 3=FWD_SEEK, 4=REV_SEEK, 0xFF=ERROR
        self._track_uid = b"\xFF" * 8      # no track
        self._subscribed: dict[int, int] = {}  # event_id → transaction_label
        session.on_message(self._on_message)

    def on_button(self, cb) -> None:
        """Register a callback: cb(operation_id: int, press: bool)."""
        self._button_cb = cb

    async def notify_volume_changed(self, new_volume: int) -> None:
        self._current_volume = new_volume & 0x7F
        await self._fire_changed(AVRCP_EVENT_VOLUME_CHANGED,
                                 bytes([self._current_volume]))

    async def notify_playback_status_changed(self, new_status: int) -> None:
        self._current_play_status = new_status & 0xFF
        await self._fire_changed(AVRCP_EVENT_PLAYBACK_STATUS_CHANGED,
                                 bytes([self._current_play_status]))

    async def notify_track_changed(self, new_track_uid: bytes) -> None:
        if len(new_track_uid) != 8:
            raise ValueError("track UID must be 8 bytes")
        self._track_uid = bytes(new_track_uid)
        await self._fire_changed(AVRCP_EVENT_TRACK_CHANGED, self._track_uid)

    async def _fire_changed(self, event_id: int, event_params: bytes) -> None:
        label = self._subscribed.pop(event_id, None)
        if label is None:
            return    # no subscriber
        payload = encode_register_notification_response(
            ctype=AvcCType.CHANGED, event_id=event_id, event_params=event_params,
        )
        await self._session.send(AVCTPMessage(
            transaction_label=label, cr=1, ipid=False,
            profile_id=AVRCP_PROFILE_ID, payload=payload,
        ))

    async def _on_message(self, msg) -> None:
        if msg.cr != 0:
            return    # we are not expecting responses
        try:
            frame = AvcFrame.from_bytes(msg.payload)
        except ValueError:
            return
        if frame.opcode == AVRCP_PASS_THROUGH_OPCODE:
            await self._handle_pass_through(msg)
        elif frame.opcode == AVRCP_VENDOR_DEPENDENT_OPCODE:
            await self._handle_vendor_dependent(msg)
        # else: silently drop unknown opcodes

    async def _handle_pass_through(self, msg) -> None:
        pt = decode_pass_through_frame(msg.payload)
        if pt.operation_id not in self._supported:
            response = encode_pass_through_response(
                ctype=AvcCType.NOT_IMPLEMENTED,
                operation_id=pt.operation_id, state_press=pt.state_press,
            )
        else:
            if self._button_cb is not None:
                res = self._button_cb(pt.operation_id, pt.state_press)
                if asyncio.iscoroutine(res):
                    await res
            response = encode_pass_through_response(
                ctype=AvcCType.ACCEPTED,
                operation_id=pt.operation_id, state_press=pt.state_press,
            )
        await self._session.send(AVCTPMessage(
            transaction_label=msg.transaction_label, cr=1, ipid=False,
            profile_id=AVRCP_PROFILE_ID, payload=response,
        ))

    async def _handle_vendor_dependent(self, msg) -> None:
        v = decode_vendor_dependent_frame(msg.payload)
        if v.pdu_id != AVRCP_PDU_REGISTER_NOTIFICATION:
            # Plan A.3 only supports REGISTER_NOTIFICATION; rest get NOT_IMPLEMENTED.
            self._send_not_implemented(msg)
            return
        event_id = v.parameters[0]
        # Record subscription so we know which label to use when the event fires.
        self._subscribed[event_id] = msg.transaction_label
        # Build INTERIM response with current state.
        if event_id == AVRCP_EVENT_VOLUME_CHANGED:
            event_params = bytes([self._current_volume])
        elif event_id == AVRCP_EVENT_PLAYBACK_STATUS_CHANGED:
            event_params = bytes([self._current_play_status])
        elif event_id == AVRCP_EVENT_TRACK_CHANGED:
            event_params = self._track_uid
        else:
            self._send_not_implemented(msg)
            return
        response = encode_register_notification_response(
            ctype=AvcCType.INTERIM, event_id=event_id, event_params=event_params,
        )
        await self._session.send(AVCTPMessage(
            transaction_label=msg.transaction_label, cr=1, ipid=False,
            profile_id=AVRCP_PROFILE_ID, payload=response,
        ))

    def _send_not_implemented(self, msg) -> None:
        # Helper to send a NOT_IMPLEMENTED with the same opcode/operands.
        # (Async send is fine to fire-and-forget via task — but for Plan A.3
        # we just construct and queue; the caller awaits as needed.)
        pass
```

```python
    )

    p.add_argument("--target", required=True, help="Peer BD address (AA:BB:CC:DD:EE:FF)")

    from pybluehost.avctp.session import AVCTPSession
```

```python
import asyncio

import pytest

from pybluehost.avctp.constants import AVCTP_PSM
from pybluehost.avctp.session import AVCTPSession
from pybluehost.profiles.classic.avrcp import (
    AVRCP_EVENT_VOLUME_CHANGED,
    AVRCP_OPERATION_PLAY,
    AVRCPController,
    AVRCPTarget,
)
from pybluehost.stack import Stack


async def test_avrcp_play_round_trip_and_volume_notification():
    iut_stack = await Stack.virtual()
    peer_stack = await Stack.virtual()
    try:
        # Wire target on peer side.
        button_log: list = []
        peer_target_holder: list[AVRCPTarget] = []

        # The acceptor-side AVRCPTarget is constructed when the inbound AVCTP channel arrives.
        async def _accept_channel(channel):
            session = AVCTPSession(channel)
            t = AVRCPTarget(session=session, stack=peer_stack)
            t.on_button(lambda op, press: button_log.append((op, press)))
            peer_target_holder.append(t)
        peer_stack.l2cap.listen_classic_psm(AVCTP_PSM, _accept_channel)

        # Establish ACL.
        await iut_stack.gap.connect(peer_stack.local_address)
        handle = next(iter(iut_stack.l2cap._classic_handles))
        channel = await iut_stack.l2cap.connect_classic_channel(handle, psm=AVCTP_PSM)
        ctrl_session = AVCTPSession(channel)
        ctrl = AVRCPController(session=ctrl_session, stack=iut_stack)

        # Wait for the target side to attach.
        for _ in range(50):
            if peer_target_holder:
                break
            await asyncio.sleep(0.02)
        assert peer_target_holder, "AVRCPTarget did not bind to incoming AVCTP channel"

        # 1) PLAY
        await ctrl.press(AVRCP_OPERATION_PLAY, hold_seconds=0.01)
        # Button log records both press + release on the target side.
        assert (AVRCP_OPERATION_PLAY, True) in button_log
        assert (AVRCP_OPERATION_PLAY, False) in button_log

        # 2) Subscribe to VOLUME_CHANGED.
        current_vol = await ctrl.subscribe(AVRCP_EVENT_VOLUME_CHANGED)
        assert current_vol == bytes([0x7F])     # default

        # 3) Target changes volume → controller gets a CHANGED.
        received_event: list[bytes] = []
        ctrl.on_notification(AVRCP_EVENT_VOLUME_CHANGED, received_event.append)

        await peer_target_holder[0].notify_volume_changed(0x40)
        for _ in range(50):
            if received_event:
                break
            await asyncio.sleep(0.02)
        assert received_event == [bytes([0x40])]

    finally:
        await iut_stack.close()
        await peer_stack.close()
```
