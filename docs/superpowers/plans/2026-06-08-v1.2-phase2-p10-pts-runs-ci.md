>
> **Operational plan, not architectural.** Unlike P.5-P.9 which built code, P.10 is mostly running things, recording outcomes, and iterating on bugs. The "TDD" cycle here is "run PTS → record verdict → fix → re-run". Calendar time (~3-4 weeks) is dominated by iteration, not feature work.

**Goal:** Close v1.2 Phase 2 by running autoptsclient against PyBlueHost across the **full LE test-case set** (GAP / GATT / L2CAP / SMP), recording pass rates, fixing the bugs PTS exposes, hooking autoptsclient into CI (over `Stack.virtual()`) as a permanent plumbing self-check, and declaring v1.2 complete in STATUS.md.

**Architecture:** No new code architecture. We introduce:
- **A status matrix**: machine-parseable YAML in `docs/pts/results/matrix/` recording per-test-case verdict + bug references. Lets us track regression and progress.
- **A CI workflow**: `.github/workflows/pts-virtual.yml` runs autoptsclient against PyBlueHost on a virtual transport. No PTS dongle in CI — we self-check BTP plumbing, not real conformance. Real conformance stays a manual operator step.
- **A pass-rate report**: rendered from the status matrix, committed at the end of each iteration to `docs/pts/results/SUMMARY.md`.

**Tech Stack:** Bash + Python for the result-matrix tooling, GitHub Actions YAML for CI. No new runtime dependencies.

```python
"""PTS test-case status-matrix tooling.

Read/update per-group YAML; render Markdown summary.

YAML schema (per group):
    group: GAP|GATT|L2CAP|SMP
    last_updated: YYYY-MM-DD
    runs: int
    cases:
      <case_id>:
        verdict: untested|pass|fail|inconc|blocked
        last_run: YYYY-MM-DD | null
        notes: str
        bug_refs: [commit_hash_or_issue_id, ...]
"""
from __future__ import annotations

import sys
from pathlib import Path
from typing import Iterable

import yaml


VALID_VERDICTS = frozenset({"untested", "pass", "fail", "inconc", "blocked"})


def load_matrix(path: str | Path) -> dict:
    return yaml.safe_load(Path(path).read_text())


def save_matrix(path: str | Path, matrix: dict) -> None:
    Path(path).write_text(yaml.safe_dump(matrix, sort_keys=False))


def update_case(
    matrix: dict, case_id: str, *,
    verdict: str | None = None,
    last_run: str | None = None,
    notes: str | None = None,
    bug: str | None = None,
) -> None:
    if verdict is not None and verdict not in VALID_VERDICTS:
        raise ValueError(
            f"invalid verdict {verdict!r}; must be one of {sorted(VALID_VERDICTS)}"
        )
    cases = matrix.setdefault("cases", {})
    case = cases.setdefault(case_id, {
        "verdict": "untested", "last_run": None, "notes": "", "bug_refs": [],
    })
    if verdict is not None:
        case["verdict"] = verdict
    if last_run is not None:
        case["last_run"] = last_run
    if notes is not None:
        case["notes"] = notes
    if bug is not None:
        case.setdefault("bug_refs", []).append(bug)


def summary_for(path: str | Path) -> dict[str, int]:
    matrix = load_matrix(path)
    counts = {v: 0 for v in VALID_VERDICTS}
    counts["total"] = 0
    for case in matrix.get("cases", {}).values():
        counts["total"] += 1
        counts[case.get("verdict", "untested")] = counts.get(
            case.get("verdict", "untested"), 0
        ) + 1
    return counts


def init_matrix(group: str, case_ids: Iterable[str]) -> dict:
    return {
        "group": group, "last_updated": None, "runs": 0,
        "cases": {
            cid: {"verdict": "untested", "last_run": None, "notes": "", "bug_refs": []}
            for cid in case_ids
        },
    }


def render_summary(matrix_paths: list[Path], output_path: Path) -> None:
    lines: list[str] = ["# PTS Auto-pts Run Summary", ""]
    lines.append("| Group | Pass | Fail | Inconc | Blocked | Untested | Total | Pass-rate |")
    lines.append("|---|---|---|---|---|---|---|---|")
    for path in sorted(matrix_paths):
        m = load_matrix(path)
        s = summary_for(path)
        total = s["total"]
        rate = f"{100 * s['pass'] // max(total, 1)}%" if total else "—"
        lines.append(
            f"| {m['group']} | {s['pass']} | {s['fail']} | {s['inconc']} | "
            f"{s['blocked']} | {s['untested']} | {total} | {rate} |"
        )
    output_path.write_text("\n".join(lines) + "\n")


def main(argv: list[str]) -> int:
    if len(argv) < 2:
        print("usage: pts_matrix.py [load|update|summary|render-summary|init] ...", file=sys.stderr)
        return 2
    cmd = argv[1]
    if cmd == "summary":
        path = Path(argv[2])
        for k, v in summary_for(path).items():
            print(f"{k}: {v}")
        return 0
    if cmd == "render-summary":
        out = Path(argv[-1])
        ins = [Path(p) for p in argv[2:-1]]
        render_summary(ins, out)
        print(f"wrote {out}")
        return 0
    print(f"unknown command: {cmd}", file=sys.stderr)
    return 2


if __name__ == "__main__":
    sys.exit(main(sys.argv))
```

**Insight:** A full autoptsclient CI run requires an autoptsserver on a Windows host with a PTS dongle. CI runners don't have that. What we **can** run in CI is:
- Spawn `pybluehost app pts-tester --transport=virtual`
- Open a TCP client (no autoptsclient, just plain Python)
- Walk through a smoke sequence: READY → Core Read Supported Services → GAP Set Powered → GAP Start Advertising → expect success
- This catches BTP plumbing regressions (the bulk of P.9 + early P.10 fixes)

This is what "CI 上 BTP 自检常态绿" (spec §11.8 P.10 success criterion) means in practice.

```python
"""CI smoke: spawn pts-tester locally, walk through a Core+GAP+GATT command sequence."""
from __future__ import annotations

import asyncio
import socket
import subprocess
import sys
from pathlib import Path

from pybluehost.pts.btp import (
    BTP_HEADER_SIZE, BtpFrame, decode_btp_frame, encode_btp_frame, opcodes as op,
)


async def _read_frame(reader: asyncio.StreamReader) -> BtpFrame:
    header = await reader.readexactly(BTP_HEADER_SIZE)
    data_len = int.from_bytes(header[3:5], "little")
    body = await reader.readexactly(data_len) if data_len else b""
    return decode_btp_frame(header + body)


async def _smoke(port: int) -> int:
    """Returns 0 on success, nonzero on failure."""
    reader, writer = await asyncio.open_connection("127.0.0.1", port)
    try:
        # READY event
        ready = await asyncio.wait_for(_read_frame(reader), timeout=10.0)
        if ready.service != op.SERVICE_CORE or ready.opcode != op.OP_CORE_EVENT_READY:
            print(f"smoke: unexpected first frame {ready}", file=sys.stderr)
            return 1

        # Core Read Supported Services
        writer.write(encode_btp_frame(BtpFrame(
            service=op.SERVICE_CORE, opcode=op.OP_CORE_READ_SUPPORTED_SERVICES,
            controller_index=op.CONTROLLER_INDEX_NONE, data=b"",
        )))
        await writer.drain()
        resp = await asyncio.wait_for(_read_frame(reader), timeout=5.0)
        if resp.data[0] != op.BTP_STATUS_SUCCESS:
            print(f"smoke: Core Read Supported Services failed: {resp.data[0]}", file=sys.stderr)
            return 1

        # GAP Set Powered = 1
        writer.write(encode_btp_frame(BtpFrame(
            service=op.SERVICE_GAP, opcode=op.OP_GAP_SET_POWERED,
            controller_index=0, data=bytes([1]),
        )))
        await writer.drain()
        resp = await asyncio.wait_for(_read_frame(reader), timeout=5.0)
        if resp.data[0] != op.BTP_STATUS_SUCCESS:
            print(f"smoke: GAP Set Powered failed: {resp.data[0]}", file=sys.stderr)
            return 1

        print("smoke: PASS (Core READY + Read Supported Services + GAP Set Powered)")
        return 0
    finally:
        writer.close()
        try:
            await writer.wait_closed()
        except Exception:
            pass


async def main() -> int:
    # Pick a free port.
    s = socket.socket(); s.bind(("127.0.0.1", 0)); _, port = s.getsockname(); s.close()

    proc = subprocess.Popen(
        [sys.executable, "-m", "pybluehost", "app", "pts-tester",
         "-t", "virtual", f"--listen=127.0.0.1:{port}"],
        stdout=subprocess.PIPE, stderr=subprocess.PIPE,
    )
    # Give it time to bind.
    for _ in range(50):
        try:
            s = socket.socket()
            s.settimeout(0.2)
            s.connect(("127.0.0.1", port))
            s.close()
            break
        except OSError:
            await asyncio.sleep(0.1)
    else:
        print("smoke: tester didn't start listening", file=sys.stderr)
        proc.terminate(); return 1

    try:
        return await _smoke(port)
    finally:
        proc.terminate()
        try:
            proc.wait(timeout=5)
        except subprocess.TimeoutExpired:
            proc.kill()


if __name__ == "__main__":
    sys.exit(asyncio.run(main()))
```
